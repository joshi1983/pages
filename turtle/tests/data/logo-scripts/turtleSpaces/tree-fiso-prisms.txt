TO tree3d
  cs pu setpos [0 80]
  ;cs = clearscreen
  ;pu = penup
  setfc 4
  ;setfc = setfillcolor
  repeat 6 [
    setfs -6 + 3 * repcount
    ;setfs = setfillshade
    bk 4 * repcount
    ;bk = back
    tent 8 * repcount 5 * repcount 10
    lo 0.1
    ;lo = lower
  ]
  setfc 8 setfs 5
  bk 50 sl 10
  ;sl = slideleft
  voxeloid 20 50 10
  ;voxeloids are stretched cubes
