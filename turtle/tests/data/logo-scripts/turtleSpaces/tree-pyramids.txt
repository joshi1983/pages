TO pyramidtree
  cs pu setpos [0 80]
  setfc 4 up 90
  repeat 6 [
    setfs -6 + 3 * repcount ra 4 * repcount
    cone 8 * repcount 5 * repcount 4
    ;while there is also a pyramid primitive, you can
    ;also create a pyramid by creating a 4-sided cone
  ]
  setfc 8 setfs 5 ra 50
  cylinder 8 50 4
END
