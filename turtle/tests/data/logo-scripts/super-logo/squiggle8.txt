TO SQUIGGLE
	FD 7
	REPEAT 8
		(FD 4 RT 45)
	FD 7
	REPEAT 8
		(FD 4 LT 45)
	FD 7
END

TO SQUIGGLE8 :SIDE :ANGLE
	REPEAT 360 / :ANGLE
		(REPEAT :SIDE (SQUIGGLE)
		RT : ANGLE)
	REPEAT 360 / :ANGLE
		(REPEAT :SIDE (SQUIGGLE)
		LT :ANGLE)
END

SQUIGGLE8 1 20
