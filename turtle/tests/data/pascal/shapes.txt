// repeat..until: draw a circle (approx.)
// 36 * 10 degree turns = 360 degrees
pencolor('red');
var step := 0;
repeat
  forward(8);
  left(10);
  step := step + 1;
until step >= 36;

// while: draw a triangle
pencolor('green');
var sides := 0;
while sides < 3 do
begin
  forward(160);
  right(120);
  sides := sides + 1;
end;

// for: draw a hexagon
pencolor('blue');
for var h := 1 to 6 do
begin
  forward(120);
  right(60);
