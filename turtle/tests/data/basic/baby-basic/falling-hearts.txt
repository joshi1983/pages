Number x = 0
Number y = 0
Number i = 0
Number p = 50
Number j = 1
Number w = 640
Number h = 480
Number x1 = 80
Number y1 = 0
Number x2 = 250
Number y2 = -150
Number x3 = 400
Number y3 = -300
Number x4 = 520
Number y4 = -450
LOOP
  Window.SetGameBackground("rgb(60,0,0)")
  ' HEART 1
  Window.SetPenColor("rgb(220,200,0)")
  i = 0
  j = 1
  While i < 1.75
    j = 0 - j
    x = x1 + w / 16 + p * j * i
    y = y1 + h / 16 + p * (Math.SquareRoot(Math.Abs(Math.Cos(i))) * Math.Cos(313 * i) - Math.SquareRoot(i))
    Window.DrawLine(x, y, x + 1, y + 1)
    i = i + 0.002
  EndWhile
  y1 = y1 + 2
  If y1 > 500 Then
    y1 = -50
    x1 = 50 + Math.GetRandomNumber(500)
  EndIf
  ' HEART 2
  Window.SetPenColor("rgb(255,100,100)")
  i = 0
  j = 1

    i = i + 0.002
  EndWhile
  y2 = y2 + 2
  If y2 > 500 Then
    y2 = -50
    x2 = 50 + Math.GetRandomNumber(500)
  EndIf
  ' HEART 3
  Window.SetPenColor("rgb(100,200,255)")
  i = 0
  j = 1
  While i < 1.75
    j = 0 - j
    x = x3 + w / 16 + p * j * i
    y = y3 + h / 16 + p * (Math.SquareRoot(Math.Abs(Math.Cos(i))) * Math.Cos(313 * i) - Math.SquareRoot(i))
    Window.DrawLine(x, y, x + 1, y + 1)
    i = i + 0.002
  EndWhile
  y3 = y3 + 2
  If y3 > 500 Then
    y3 = -50
    x3 = 50 + Math.GetRandomNumber(500)
  EndIf
  ' HEART 4
  Window.SetPenColor("rgb(150,255,150)")
  i = 0
  j = 1
  While i < 1.75
    j = 0 - j
    x = x4 + w / 16 + p * j * i
    y = y4 + h / 16 + p * (Math.SquareRoot(Math.Abs(Math.Cos(i))) * Math.Cos(313 * i) - Math.SquareRoot(i))

    i = i + 0.002
  EndWhile
  y4 = y4 + 2

    y4 = -50
    x4 = 50 + Math.GetRandomNumber(500)
  EndIf
  Window.SetPenColor("rgb(230,220,200)")
  Window.DrawText("4 hearts", 10, 20, 14)
END LOOP
