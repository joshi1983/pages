' Baby in Diapers - Pixel Art
' Shows how DrawSolidRectangle(1,1) works as SetPixel
Number originX = 300      ' baby start X
Number originY = 80       ' baby start Y
Number pixelSize = 8      ' pixel size
LOOP
    Window.SetGameBackground("black")
    ' Title text above hands
    Window.SetBrushColor("yellow")
    Window.DrawText("baby basic programming", originX - 30, originY - 25, 20)
    ' Skin color
    Window.SetBrushColor("#FFDBAC")
    ' Hands up
    Window.DrawSolidRectangle(originX, originY, pixelSize*2, pixelSize*2)         ' left hand
    Window.DrawSolidRectangle(originX+80, originY, pixelSize*2, pixelSize*2)      ' right hand
    ' Arms raised
    Window.DrawSolidRectangle(originX+4, originY+16, pixelSize, pixelSize*5)     ' left forearm
    Window.DrawSolidRectangle(originX+12, originY+50, pixelSize*3, pixelSize)    ' left shoulder
    Window.DrawSolidRectangle(originX+80, originY+16, pixelSize, pixelSize*5)    ' right forearm
    Window.DrawSolidRectangle(originX+52, originY+50, pixelSize*3, pixelSize)    ' right shoulder
    ' Head
    Window.DrawSolidRectangle(originX+20, originY+30, pixelSize*6, pixelSize*5)   ' head
    ' Eyes and smile - black pixels
    Window.SetBrushColor("black")
    Window.DrawSolidRectangle(originX+28, originY+40, pixelSize, pixelSize)      ' left eye
    Window.DrawSolidRectangle(originX+52, originY+40, pixelSize, pixelSize)      ' right eye
    Window.DrawSolidRectangle(originX+36, originY+56, pixelSize*2, pixelSize)    ' smile
    ' Body
    Window.SetBrushColor("#FFDBAC")
    Window.DrawSolidRectangle(originX+20, originY+70, pixelSize*6, pixelSize*4)   ' body
    ' Diaper - white
    Window.SetBrushColor("white")
    Window.DrawSolidRectangle(originX+12, originY+80, pixelSize*8, pixelSize*4)   ' diaper
    ' Legs
    Window.SetBrushColor("#FFDBAC")
    Window.DrawSolidRectangle(originX+20, originY+112, pixelSize*2, pixelSize*3)  ' left leg
    Window.DrawSolidRectangle(originX+52, originY+112, pixelSize*2, pixelSize*3)  ' right leg
END LOOP
