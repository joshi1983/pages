var x,y,i,p,j,w,h,x1,y1
mode 1: wcolor 250,250,250 : fcolor 190,10,0
w=800 : h=600 : p=50 :j=1

x1 = 400: y1 = 300

i=0
p = 150
while i < 1.75
	j = -j
	x = x1 + w/16+p*j*i
	y = y1 + h/16+p*(sqr(cos(i)) *cos(313*i)-sqr(i))
	frect x,y,2,2
	i=i+0.0004
wend
swap
